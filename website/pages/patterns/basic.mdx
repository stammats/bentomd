footer: true

\`\`\`typescript
import { z } from 'zod';
import { router, publicProcedure } from './trpc';

const CreateUserSchema = z.object({
  name: z.string().min(2).max(100),
  email: z.string().email(),
  role: z.enum(['admin', 'member', 'viewer']),
});

export const userRouter = router({
  create: publicProcedure
    .input(CreateUserSchema)
    .mutation(async ({ input, ctx }) => {
      // input is fully typed as { name: string; email: string; role: ... }
      const user = await ctx.db.user.create({ data: input });
      return { id: user.id, ...input };
    }),
});
\`\`\``} />

  method: 'POST',

  body: JSON.stringify({
    name: 'Alice',
    email: 'alice@example.com',
    role: 'admin',
  }),
});

// Hope the response shape is correct
const user = await res.json();
console.log(user.id); // any

// Fully typed, validated, autocompleted
const user = await trpc.user.create.mutate({

// TypeScript knows the exact shape
console.log(user.id); // string

## Bento Grid

A flexible grid layout where each cell can span multiple columns and rows. Cells support icon, title, description, image, value, and label. Use the `size` keyword for quick sizing or explicit `"CxR"` notation.

### Size keywords

| Keyword | Columns | Rows | Description |
|---|---|---|---|
| `sm` | 4 (1/3) | 1 | Small card |
| `md` | 6 (1/2) | 1 | Medium card |
| `lg` | 8 (2/3) | 1 | Large card |
| `wide` | 12 (full) | 1 | Full-width banner |
| `tall` | 4 (1/3) | 2 | Tall sidebar card |
| `hero` | 8 (2/3) | 2 | Hero feature block |
| `full` | 12 (full) | 2 | Full-width hero |

You can also use explicit notation: `size: "6x2"` means 6 columns wide, 2 rows tall.
